#include <bits/stdc++.h>
#include <atcoder/modint>
using i64 = long long;
using P = std::pair<int, int>;

int main() {
  std::cin.tie(nullptr)->sync_with_stdio(false);
  i64 n, m;
  std::cin >> n >> m;
  std::vector<i64> a(n), pref(n + 1);
  for (int i = 0; i < n; ++i) {
    std::cin >> a[i];
    pref[i + 1] = pref[i] + a[i];
  }

  i64 sum = 0, ans = 0;
  for (i64 i = 0; i < m - 1; ++i) sum += (i + 1) * a[i];
  ans = sum + m * a[m - 1];
  for (i64 i = m - 1; i < n; ++i) {
    sum += m * a[i];
    ans = std::max(ans, sum);
    sum -= pref[i + 1] - pref[i - m + 1];
  }

  std::cout << ans << "\n";
}
