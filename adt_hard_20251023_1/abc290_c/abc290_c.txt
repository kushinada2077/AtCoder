#include <bits/stdc++.h>
using i64 = long long;
using P = std::pair<int, int>;

int main() {
  std::cin.tie(nullptr)->sync_with_stdio(false);
  int n, k;
  std::cin >> n >> k;
  std::vector<int> a(n);
  for (int i = 0; i < n; ++i) {
    std::cin >> a[i];
  }

  std::ranges::sort(a);
  a.erase(std::unique(a.begin(), a.end()), a.end());
  int ans = 0;
  for (int i = 0; i < k; ++i) {
    if (ans != a[i]) break;
    ans++;
  }

  std::cout << ans << "\n";
}
