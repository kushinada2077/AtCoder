#include <bits/stdc++.h>
using i64 = long long;
using P = std::pair<int, int>;

int main() {
  std::cin.tie(nullptr)->sync_with_stdio(false);
  int n;
  std::cin >> n;
  std::vector<bool> chk(2 * n + 2);
  int cnt = 0;

  auto make_num = [&]() {
    for (int i = 1; i < 2 * n + 2; ++i)
      if (chk[i] == 0) return i;
    return 0;
  };

  while (cnt < n) {
    int x = make_num();
    chk[x] = true;
    std::cout << x << std::endl;
    std::cin >> x;
    chk[x] = true;
    if (x == 0) break;
  }
}
