#include <bits/stdc++.h>
using i64 = long long;
using P = std::pair<int, int>;
using P_i64 = std::pair<i64, i64>;

int main() {
  std::cin.tie(nullptr)->sync_with_stdio(false);
  int n;
  std::cin >> n;
  std::vector<i64> re(13);
  re[1] = 1;
  for (int i = 2; i < 13; ++i) {
    re[i] = 10 * re[i - 1] + 1;
  }

  std::vector<i64> ans;

  for (int i = 1; i < 13; ++i) {
    for (int j = 1; j < 13; ++j) {
      for (int k = 1; k < 13; ++k) {
        ans.push_back(re[i] + re[j] + re[k]);
      }
    }
  }

  std::ranges::sort(ans);
  ans.erase(std::unique(ans.begin(), ans.end()), ans.end());
  std::cout << ans[n - 1] << "\n";
}
