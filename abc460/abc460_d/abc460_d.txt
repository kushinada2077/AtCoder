#include <bits/stdc++.h>

#ifndef ONLINE_JUDGE
#define kushinada freopen(getenv("MY_PATH"), "r", stdin);
#else
#define kushinada
#endif

using i64 = long long;
using P = std::pair<int, int>;
using TP = std::tuple<int, int, int>;

int main() {
  std::cin.tie(NULL)->sync_with_stdio(false);
  kushinada;
  const int INF = 0x3f3f3f3f - 1;
  int H, W;
  std::cin >> H >> W;
  std::vector<std::string> S(H);
  std::queue<P> que;
  std::vector dis(H, std::vector<int>(W, INF));
  for (int i = 0; i < H; ++i) {
    std::cin >> S[i];
  }

  std::vector dy = {-1, -1, 0, 1, 1, 1, 0, -1};
  std::vector dx = {0, 1, 1, 1, 0, -1, -1, -1};
  auto chk = [&](int y, int x) -> bool {
    for (int dir = 0; dir < 8; ++dir) {
      int ny = y + dy[dir];
      int nx = x + dx[dir];
      if (ny < 0 || ny >= H || nx < 0 || nx >= W) {
        continue;
      }
      if (S[ny][nx] == '#') {
        return true;
      }
    }
    return false;
  };

  std::vector NS(H, std::vector<char>(W));
  for (int i = 0; i < H; ++i) {
    for (int j = 0; j < W; ++j) {
      if (S[i][j] == '#') {
        NS[i][j] = '.';
      } else {
        NS[i][j] = chk(i, j) ? '#' : '.';
      }

      if (NS[i][j] == '#') {
        dis[i][j] = 0;
        que.emplace(i, j);
      }
    }
  }

  while (!que.empty()) {
    auto [y, x] = que.front();
    que.pop();
    for (int dir = 0; dir < 8; ++dir) {
      int ny = y + dy[dir];
      int nx = x + dx[dir];
      if (ny < 0 || ny >= H || nx < 0 || nx >= W) {
        continue;
      }
      if (NS[ny][nx] == '#' || dis[ny][nx] != INF) {
        continue;
      }
      dis[ny][nx] = dis[y][x] + 1;
      que.emplace(ny, nx);
    }
  }

  for (int i = 0; i < H; ++i) {
    for (int j = 0; j < W; ++j) {
      if (dis[i][j] % 2) {
        std::print("#");
      } else {
        std::print(".");
      }
    }
    std::println();
  }
}
