#include <bits/stdc++.h>
#include <atcoder/modint>
using i64 = long long;
using P = std::pair<int, int>;

int divisor_cnt(int x) {
  int cnt = 0;
  for (int i = 1; i * i <= x; ++i) {
    if (x % i == 0) cnt += 2;
    if (i * i == x) cnt--;
  }
  return cnt;
}

int main() {
  std::cin.tie(nullptr)->sync_with_stdio(false);
  int n;
  std::cin >> n;
  i64 ans = 0;
  for (int a = 1; a <= n; ++a) {
    for (int b = 1; a * b <= n; ++b) {
      ans += divisor_cnt(n - a * b);
    }
  }

  std::cout << ans << "\n";
}
