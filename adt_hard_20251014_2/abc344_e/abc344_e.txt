#include <bits/stdc++.h>
#include <atcoder/modint>
using i64 = long long;
using P = std::pair<int, int>;

int main() {
  std::cin.tie(nullptr)->sync_with_stdio(false);
  int n;
  std::cin >> n;
  std::list<int> a;
  std::map<int, std::list<int>::iterator> hash;
  for (int i = 0; i < n; ++i) {
    int x;
    std::cin >> x;
    a.push_back(x);
  }

  for (auto it = a.begin(); it != a.end(); ++it) {
    hash[*it] = it;
  }

  int q;
  std::cin >> q;
  for (int i = 0; i < q; ++i) {
    int type;
    std::cin >> type;
    if (type == 1) {
      int x, y;
      std::cin >> x >> y;
      auto it = hash[x];
      it++;
      hash[y] = a.insert(it, y);
    } else if (type == 2) {
      int x;
      std::cin >> x;
      auto it = hash[x];
      a.erase(it);
      hash.erase(x);
    }
  }

  for (auto x : a) std::cout << x << " ";
}
