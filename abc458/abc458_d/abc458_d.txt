using i64 = long long;
using P = std::pair<int, char>;
using TP = std::tuple<int, int, int>;

int main() {
  std::cin.tie(NULL)->sync_with_stdio(false);
  kushinada;
  int X, Q;
  std::cin >> X >> Q;
  std::priority_queue<int> l;
  std::priority_queue<int, std::vector<int>, std::greater<int>> r;
  l.push(X);
  for (int i = 0; i < Q; ++i) {
    int A, B;
    std::cin >> A >> B;
    if (A > B) std::swap(A, B);
    if (B < l.top()) {
      l.push(A);
      l.push(B);
      r.push(l.top());
      l.pop();
    } else if (A > l.top()) {
      r.push(A);
      r.push(B);
      l.push(r.top());
      r.pop();
    } else {
      l.push(A);
      r.push(B);
    }
    std::println("{}", l.top());
  }
}
