  kushinada;
  int X, Q;
  std::cin >> X >> Q;
  std::priority_queue<int> l;
  std::priority_queue<int, std::vector<int>, std::greater<int>> r;
  l.push(X);
  auto add = [&](int x) -> void {
    if (l.top() < x) r.push(x);
    else l.push(x);
  };
  for (int i = 0; i < Q; ++i) {
    int A, B;
    std::cin >> A >> B;
    add(A), add(B);
    if (l.size() + 1 == r.size()) {
      l.push(r.top());
      r.pop();
    } else if (r.size() + 3 == l.size()) {
      r.push(l.top());
      l.pop();
    }
    std::println("{}", l.top());
  }
}
