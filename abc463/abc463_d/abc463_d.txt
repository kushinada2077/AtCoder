#include <bits/stdc++.h>

#ifndef ONLINE_JUDGE
#define kushinada freopen(getenv("MY_PATH"), "r", stdin);
#else
#define kushinada
#endif

using i64 = long long;
using P = std::pair<int, int>;
using TP = std::tuple<int, int, int>;

int main() {
  std::cin.tie(NULL)->sync_with_stdio(false);
  kushinada;
  int n, k;
  std::cin >> n >> k;
  std::vector<P> lr(n);
  for (auto& [l, r] : lr) {
    std::cin >> l >> r;
  }

  std::ranges::sort(lr, {}, &P::second);

  int ans{*std::ranges::partition_point(std::views::iota(0, 0x3f3f3f3f), [k, lr](int x) -> bool {
    int last{}, chosen{};

    for (auto [l, r] : lr) {
      if (last <= l) {
        chosen++;
        last = r + x + 1;
      }
    }

    return chosen >= k;
  })};

  if (ans) {
    std::println("{}", ans);
  } else {
    std::println("-1");
  }
}
