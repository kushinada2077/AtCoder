using i64 = long long;
using P = std::pair<int, int>;
using TP = std::tuple<int, int, int>;

int main() {
  std::cin.tie(NULL)->sync_with_stdio(false);
  kushinada;
  int n;
  std::cin >> n;
  std::vector<P> taka;
  for (int i = 0; i < n; ++i) {
    int h, l;
    std::cin >> h >> l;
    while (!taka.empty() && taka.back().second <= h) {
      taka.pop_back();
    }
    taka.emplace_back(l, h);
  }

  int q;
  std::cin >> q;
  for (int i = 0; i < q; ++i) {
    int t;
    std::cin >> t;
    auto x = std::ranges::upper_bound(taka, t, {}, &P::first)->second;
    std::println("{}", x);
  }
}
