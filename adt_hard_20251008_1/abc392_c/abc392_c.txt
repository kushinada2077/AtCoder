#include <bits/stdc++.h>
using i64 = long long;
using P = std::pair<int, int>;

int main() {
  std::cin.tie(nullptr)->sync_with_stdio(false);
  int n;
  std::cin >> n;
  std::vector<int> p(n), q(n), r(n);
  for (int i = 0; i < n; ++i) {
    std::cin >> p[i];
    p[i]--;
  }
  for (int i = 0; i < n; ++i) {
    std::cin >> q[i];
    q[i]--;
    r[q[i]] = i;
  }

  for (int i = 0; i < n; ++i) {
    std::cout << q[p[r[i]]] + 1 << " \n"[i == n - 1];
  }
}
