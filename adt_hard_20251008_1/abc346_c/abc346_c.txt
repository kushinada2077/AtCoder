#include <bits/stdc++.h>
using i64 = long long;
using P = std::pair<int, int>;

int main() {
  std::cin.tie(nullptr)->sync_with_stdio(false);
  i64 n, k;
  std::cin >> n >> k;
  std::set<i64> a;
  for (int i = 0; i < n; ++i) {
    i64 x;
    std::cin >> x;
    a.insert(x);
  }

  i64 ans = (1LL + k) * k / 2;

  for (auto x : a) {
    if (x <= k) ans -= x;
  }
  std::cout << ans << "\n";
}
