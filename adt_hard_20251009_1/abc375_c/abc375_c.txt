#include <bits/stdc++.h>
#include <atcoder/modint>
using i64 = long long;
using P = std::pair<int, int>;

int main() {
  std::cin.tie(nullptr)->sync_with_stdio(false);
  int n;
  std::cin >> n;
  std::vector<std::string> board(n);
  std::vector cnt(n, std::vector<int>(n, 0));
  for (int i = 0; i < n; ++i) {
    std::cin >> board[i];
  }

  for (int x = 0; x < n; ++x) {
    for (int y = 0; y < n; ++y) {
      cnt[x][y] = (std::min({x, y, n - x - 1, n - y - 1}) + 1) % 4;
    }
  }

  std::vector ans(n, std::vector<char>(n));
  for (int i = 0; i < n; ++i) {
    for (int j = 0; j < n; ++j) {
      int y = i, x = j;
      for (int k = 0; k < cnt[i][j]; ++k) {
        int ny = x;
        int nx = n - 1 - y;
        y = ny;
        x = nx;
      }
      ans[y][x] = board[i][j];
    }
  }

  for (int i = 0; i < n; ++i) {
    for (int j = 0; j < n; ++j) {
      std::cout << ans[i][j];
    }
    std::cout << "\n";
  }
}
