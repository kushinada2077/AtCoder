#include <bits/stdc++.h>
#include <atcoder/modint>
using i64 = long long;
using P = std::pair<int, int>;

int main() {
  std::cin.tie(nullptr)->sync_with_stdio(false);
  int n;
  std::cin >> n;
  std::map<std::string, int> hash;
  for (int i = 0; i < n; ++i) {
    std::string s;
    std::cin >> s;
    if (hash[s] == 0) std::cout << s << "\n";
    else std::cout << s << "(" << hash[s] << ")" << "\n";
    hash[s]++;
  }
}
