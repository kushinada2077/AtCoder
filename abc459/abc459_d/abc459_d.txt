#include <bits/stdc++.h>

#ifndef ONLINE_JUDGE
#define kushinada freopen(getenv("MY_PATH"), "r", stdin);
#else
#define kushinada
#endif

using i64 = long long;
using P = std::pair<int, char>;
using TP = std::tuple<int, int, int>;

int main() {
  std::cin.tie(NULL)->sync_with_stdio(false);
  kushinada;
  int T;
  std::cin >> T;
  for (int i = 0; i < T; ++i) {
    std::string S;
    std::cin >> S;
    int n = S.size();
    std::vector<int> cnt(26);
    for (auto c : S) {
      cnt[c - 'a']++;
    }

    int max = *std::ranges::max_element(cnt);
    if (2 * max > n + 1) {
      std::println("No");
      continue;
    }

    std::string ans = "";
    int pre = -1;
    for (int i = 0; i < n; ++i) {
      int max_i = -1;
      for (int j = 0; j < 26; ++j) {
        if (pre == j) continue;
        if (max_i == -1 || cnt[max_i] < cnt[j]) max_i = j;
      }
      ans += char(max_i + 'a');
      cnt[max_i]--;
      pre = max_i;
    }

    std::println("Yes\n{}", ans);
  }
}
