#include <bits/stdc++.h>

#ifndef ONLINE_JUDGE
#define kushinada freopen(getenv("MY_PATH"), "r", stdin);
#else
#define kushinada
#endif

using i64 = long long;
using P = std::pair<int, char>;
using TP = std::tuple<int, int, int>;

int main() {
  std::cin.tie(NULL)->sync_with_stdio(false);
  kushinada;
  int T;
  std::cin >> T;
  for (int i = 0; i < T; ++i) {
    std::string S;
    std::cin >> S;
    int n = S.size();
    std::vector<int> cnt(26);
    for (auto c : S) {
      cnt[c - 'a']++;
    }

    int max = *std::ranges::max_element(cnt);
    if (2 * max > n + 1) {
      std::println("No");
      continue;
    }

    std::vector<char> cha;
    for (int i = max; i > 0; --i) {
      for (int j = 0; j < 26; ++j) {
        if (cnt[j] == i) {
          for (int k = 0; k < i; ++k) {
            cha.push_back(j + 'a');
          }
        }
      }
    }

    std::vector<char> ans(n);
    int j = 0;
    for (int i = 0; i < n; ++i) {
      ans[j] = cha[i];
      j = (j + 2) % n;
      if (ans[j]) j++;
    }

    std::println("Yes");
    std::println("{}", std::string(ans.begin(), ans.end()));
  }
}
